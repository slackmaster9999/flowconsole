import {

  ExternalService,
  ComputerSystem,
  ReactApp,
  Container,
  RestApi,
  MessageQueue,
  BackgroundJob,

const employee = new User({ name: "Regional Planner", description: "Creates purchase orders" });

const portal = new ReactApp({
  name: "Planner Portal",
  description: "Next.js dashboard for procurement team",
  belongsTo: atlas
});

const integrationHub = new Container({
  name: "Integration Hub",
  description: "Async orchestrations + adapters",
  belongsTo: atlas
});

const planningApi = new RestApi({
  name: "Planning Service",
  description: "Handles demand & supply planning",
  belongsTo: integrationHub
});

const inventoryApi = new RestApi({
  name: "Inventory Service",
  description: "Tracks warehouse stock",
  belongsTo: integrationHub
});

const workflowQueue = new MessageQueue({
  name: "Workflow Queue",
  description: "Commands for async processing",
  belongsTo: integrationHub
});

const reportingJob = new BackgroundJob({
  name: "Nightly Reconciliation",
  description: "Produces compliance extracts",
  belongsTo: integrationHub
});

const erpDb = new Postgres({
  name: "ERP Database",
  description: "Orders, forecasts, contracts",
  belongsTo: integrationHub
});

employee
  .sendsRequest(portal, "create purchase order")
  .then(portal).sendsRequest(planningApi, "submit plan")
  .then(planningApi).sendsRequest(inventoryApi, "reserve stock")
  .then(planningApi).getDataFrom(erpDb, "fetch demand")
  .then(inventoryApi).getDataFrom(erpDb, "current stock")
  .then(planningApi).sendsRequest(workflowQueue, "publish workflow")
  .then(planningApi).sendsRequest(supplier, "send order");

reportingJob
  .sendsRequest(erpDb, "load data")
  .executesRequest("generate nightly reports")
  .sendsRequest(workflowQueue, "notify portal");

import flowconsole.sdk.*;

public class GlobalErpExample {
  public static void main(String[] args) {
    User employee = User.Builder.create()
      .name("Regional Planner")
      .description("Creates purchase orders")
      .build();

    ExternalService supplier = ExternalService.Builder.create()
      .name("Supplier API")
      .description("Partner integration")
      .build();

    ComputerSystem atlas = ComputerSystem.Builder.create()
      .name("Atlas ERP")
      .build();

    ReactApp portal = ReactApp.Builder.create()
      .name("Planner Portal")
      .description("Next.js dashboard for procurement team")
      .belongsTo(atlas)
      .build();

    Container integrationHub = Container.Builder.create()
      .name("Integration Hub")
      .description("Async orchestrations + adapters")
      .belongsTo(atlas)
      .build();

    RestApi planningApi = RestApi.Builder.create()
      .name("Planning Service")
      .description("Handles demand & supply planning")
      .belongsTo(integrationHub)
      .build();

    RestApi inventoryApi = RestApi.Builder.create()
      .name("Inventory Service")
      .description("Tracks warehouse stock")
      .belongsTo(integrationHub)
      .build();

    MessageQueue workflowQueue = MessageQueue.Builder.create()
      .name("Workflow Queue")
      .description("Commands for async processing")
      .belongsTo(integrationHub)
      .build();

    BackgroundJob reportingJob = BackgroundJob.Builder.create()
      .name("Nightly Reconciliation")
      .description("Produces compliance extracts")
      .belongsTo(integrationHub)
      .build();

    Postgres erpDb = Postgres.Builder.create()
      .name("ERP Database")
      .description("Orders, forecasts, contracts")
      .belongsTo(integrationHub)
      .build();

    employee
      .sendsRequest(portal, "create purchase order")
      .then(portal).sendsRequest(planningApi, "submit plan")
      .then(planningApi).sendsRequest(inventoryApi, "reserve stock")
      .then(planningApi).getDataFrom(erpDb, "fetch demand")
      .then(inventoryApi).getDataFrom(erpDb, "current stock")
      .then(planningApi).sendsRequest(workflowQueue, "publish workflow")

    reportingJob
      .sendsRequest(erpDb, "load data")
      .executesRequest("generate nightly reports")
      .sendsRequest(workflowQueue, "notify portal");
