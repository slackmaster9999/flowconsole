```typescript scroll=false showLineNumbers
import {
  User,
  ComputerSystem,
  Container,
  ReactApp,
  RestApi,
  Redis,
  Postgres,
  KafkaTopic,
  ExternalService
} from "@flowconsole/sdk";

const user = new User({ name: "Customer", description: "Retail banking customer" });

const system = new ComputerSystem({ name: "Cloud Banking" });
const storage = new Container({ name: "Data Store", belongsTo: system });
const backend = new Container({ name: "Core Services", belongsTo: system });

const frontApp = new ReactApp({
  name: "Customer Dashboard",
  description: "Browser Single-page Application",
  belongsTo: system
});

const authApi = new RestApi({
  name: "Authentication",
  description: "Self-Hosted Authentication Service",
  belongsTo: backend
});

const accountsApi = new RestApi({
  name: "Accounts API",
  description: "Java Spring service for balances and payments",
  belongsTo: backend
});

const cache = new Redis({
  name: "Session Cache",
  description: "Stores customer session payloads",
  belongsTo: storage
});

const db = new Postgres({
  name: "Ledger DB",
  description: "Persistent customer and transaction data",
  belongsTo: storage
});

const auditStream = new KafkaTopic({
  name: "Audit Events",
  description: "Every request produces an audit record",
  belongsTo: backend
});

const fraudService = new ExternalService({
  name: "Fraud Guard",
  description: "3rd-party fraud scoring",
  belongsTo: backend
});

user
  .sendsRequest(frontApp, "launch app")
  .then(frontApp).sendsRequest(authApi, "login")
  .then(frontApp).sendsRequest(accountsApi, "load dashboard")
  .then(accountsApi).getDataFrom(cache, "session context")
  .then(accountsApi).getDataFrom(db, "account snapshot")
  .then(accountsApi).sendsRequest(authApi, "validate token")
  .then(accountsApi).sendsRequest(auditStream, "emit audit")
  .then(accountsApi).sendsRequest(fraudService, "score transaction");
```

      .build();
