```typescript scroll=false showLineNumbers
import {
  User,
  ComputerSystem,
  Container,
  ReactApp,
  RestApi,
  BackgroundJob,
  MessageQueue,
  KafkaTopic,
  Postgres
} from "@flowconsole/sdk";

const contributor = new User({ name: "Contributor", description: "Sends pull requests" });
const maintainer = new User({ name: "Maintainer", description: "Reviews and deploys" });

const helios = new ComputerSystem({ name: "Helios OSS" });
const gitGateway = new Container({ name: "Git Gateway", belongsTo: helios });
const ciCluster = new Container({ name: "CI Cluster", belongsTo: helios });
const observability = new Container({ name: "Observability", belongsTo: helios });

const webApp = new ReactApp({
  name: "Helios Web",
  description: "Next.js UI for issues, merge requests, pipelines",
  belongsTo: helios
});

const gitHttp = new RestApi({
  name: "Git HTTP",
  description: "Clone & push over HTTPS",
  belongsTo: gitGateway
});

const apiGateway = new RestApi({
  name: "GraphQL API",
  description: "Issues, projects, releases",
  belongsTo: gitGateway
});

const ciRunner = new BackgroundJob({
  name: "CI Runner",
  description: "Executes pipelines from queue",
  belongsTo: ciCluster
});

const pipelineQueue = new MessageQueue({
  name: "Pipeline Queue",
  description: "Jobs waiting for runners",
  belongsTo: ciCluster
});

const eventStream = new KafkaTopic({
  name: "Activity Stream",
  description: "Push events, comments, deployments",
  belongsTo: observability
});

const metricsStore = new Postgres({
  name: "Metrics Store",
  description: "Usage, billing, analytics",
  belongsTo: observability
});

contributor
  .sendsRequest(webApp, "open MR")
  .then(webApp).sendsRequest(apiGateway, "create merge request")
  .sendsRequest(gitHttp, "push commits")
  .then(apiGateway).sendsRequest(pipelineQueue, "enqueue pipeline")
  .sendsRequest(eventStream, "publish activity");

ciRunner
  .sendsRequest(pipelineQueue, "claim job")
  .executesRequest("run tests")
  .sendsRequest(apiGateway, "update status")
  .sendsRequest(eventStream, "emit pipeline events");

maintainer
  .sendsRequest(webApp, "review & deploy")
  .then(webApp).sendsRequest(apiGateway, "approve merge")
  .sendsRequest(metricsStore, "record deployment")
  .then(apiGateway).sendsRequest(eventStream, "log deployment");
```
